extends GameModule
## Outtap game screen. Owns one PuzzleSession at a time and plays its results on the BoardView.
## Flow: start → taps → clear/fail card → next (natural ad break) or free restart.
const COMBO_MILESTONES := [3, 6, 10, 15, 20, 30, 40, 50]
const CLOCK_FLUSH_MS := 5000.0

var session: PuzzleSession
var mode := "campaign"
var index := 1
var daily_date := ""

var _root: Control
var _board_view: BoardView
var _hearts: HeartsView
var _practice: PanelContainer
var _title: Label
var _subtitle: Label
var _coach: PanelContainer
var _coach_label: Label
var _coach_action: BaseButtonView
var _hint_button: ToolButton
var _undo_button: ToolButton
var _restart_button: ToolButton
var _overlay: GameOverlay
var _continue_button: BaseButtonView
var _continue_note: Label

var _clock_ms := 0.0
var _paused := false
var _ended := false
var _transitioning := false
var _ad_busy := false
var _retries := {}
var _coach_key := ""

func _ready() -> void:
	add_to_group("outtap_game")
	_build()
	AppManager.game_back_handler = _handle_back
	AdsManager.rewarded_changed.connect(_refresh_buttons)
	AdsManager.rewarded_changed.connect(_refresh_continue)

func _exit_tree() -> void:
	if AppManager.game_back_handler == Callable(self, "_handle_back"):
		AppManager.game_back_handler = Callable()

# --- GameModule lifecycle ----------------------------------------------------------------------

func start_session() -> void:
	super.start_session()
	var request: Dictionary = Progress.request.duplicate()
	Progress.request = {}
	_start_from_request(request)

func pause_session() -> void:
	super.pause_session()
	_paused = true
	_board_view.cancel_input()
	_flush_clock()
	_save_attempt(true)

func resume_session() -> void:
	super.resume_session()
	_paused = false

func stop_session() -> void:
	super.stop_session()
	_flush_clock()
	_save_attempt(true)

func _set_coach(key: String, action_key: String = "", action: Callable = Callable()) -> void:
	_coach_key = key
	var reduced := bool(SettingsManager.get_value("reduced_motion"))
	if key.is_empty():
		var hide := create_tween()
		hide.tween_property(_coach, "modulate:a", 0.0, 0.0 if reduced else 0.18)
		return
	_coach_label.text = key
	for connection in _coach_action.pressed.get_connections():
		_coach_action.pressed.disconnect(connection["callable"])
	_coach_action.visible = action.is_valid()
	if action.is_valid():
		_coach_action.text = action_key
		_coach_action.pressed.connect(action)
	_coach.modulate.a = 0.0 if not reduced else 1.0
	var show := create_tween()
	show.tween_property(_coach, "modulate:a", 1.0, 0.0 if reduced else 0.22)

# --- level flow --------------------------------------------------------------------------------

func _start_from_request(request: Dictionary) -> void:
	var resume_data := {}
	var level := {}
	if bool(request.get("resume", false)):
		resume_data = Progress.active_attempt()
		level = Levels.by_id(str(resume_data.get("level_id", "")))
		mode = str(resume_data.get("mode", "campaign"))
		daily_date = str(resume_data.get("daily_date", ""))
	if level.is_empty():
		resume_data = {}
		mode = str(request.get("mode", "campaign"))
		match mode:
			"switch":
				index = int(request.get("index", 1))
				level = Levels.switch_level(index)
			"daily":
				daily_date = str(request.get("date", Levels.today_key()))
				level = Levels.daily_for(daily_date)
			_:
				mode = "campaign"
				index = int(request.get("index", Progress.reached()))
				level = Levels.campaign(index)
	else:
		index = Levels.index_of(str(level.get("id", "")))
	_begin_level(level, resume_data, false)

func _begin_level(level: Dictionary, resume_data: Dictionary, retry: bool) -> void:
	_ended = false
	_transitioning = false
	_overlay.close()
	_clock_ms = 0.0
	var level_id := str(level.get("id", ""))
	var restored: PuzzleSession = null
	if not resume_data.is_empty():
		restored = PuzzleSession.from_save(level, resume_data)
		if restored == null:
			Diagnostics.log_event("game", "attempt_discarded", {"level_id": level_id, "reason": "content_or_rules_changed"})
			UI.toast("ATTEMPT_RESET")
	if restored != null and restored.status != PuzzleSession.Status.CLEARED:
		session = restored
	else:
		var retries := int(_retries.get(level_id, 0))
		session = PuzzleSession.create(level, mode, _new_attempt_id(), retries)
		session.daily_date = daily_date if mode == "daily" else ""
		_track_level("level_started", {})
		if level_id == "c001" and not Progress.flag("tutorial_started"):
			Progress.set_flag("tutorial_started")
			AnalyticsService.track(&"tutorial_started", {"level_id": level_id})
	_board_view.input_enabled = true
	_board_view.setup(session.board, _style(), _link_labels(session.board), not SettingsManager.get_value("reduced_motion"))
	_update_header()
	_hearts.set_lives(session.lives, false)
	_hearts.visible = not session.soft_mistakes
	_practice.visible = session.soft_mistakes
	_undo_button.visible = session.has_undo_feature()
	_refresh_buttons()
	_show_level_coach()
	if session.hint_target != -1:
		_board_view.show_hint(session.hint_target)
	_save_attempt(true)
	if session.status == PuzzleSession.Status.FAILED:
		_show_fail_card()
	elif session.is_stuck():
		_show_stuck()

func _style() -> Dictionary:
	match mode:
		"switch": return Palette.SWITCH_STYLE
		"daily": return Palette.DAILY_STYLE
		_: return Palette.chapter_style(Levels.chapter_of(index))

## Switches get letters A, B, C...; linked blocks show the letters of every switch that turns them.
func _link_labels(board: BoardState) -> Dictionary:
	var labels := {}
	var letter := 0
	for id in board.size():
		if board.kinds[id] == BoardState.KIND_SWITCH:
			var name := String.chr(65 + letter)
			letter += 1
			labels[id] = labels.get(id, {})
			labels[id]["switch"] = name
			for target in board.links[id]:
				labels[target] = labels.get(target, {})
				labels[target]["targets"] = str(labels[target].get("targets", "")) + name
	return labels

func _update_header() -> void:
	match mode:
		"switch":
			_title.text = tr("SWITCH_N") % index
			_subtitle.text = tr("SWITCH_SUBTITLE")
		"daily":
			_title.text = tr("DAILY_TITLE")
			_subtitle.text = _format_date(daily_date)
		_:
			_title.text = tr("STAGE_N") % index
			_subtitle.text = tr("CHAPTER_N") % Levels.chapter_of(index)

func _format_date(key: String) -> String:
	var parts := key.split("-")
	if parts.size() != 3:
		return key
	return tr("DATE_FORMAT") % [int(parts[1]), int(parts[2])]

func _show_level_coach() -> void:
	var key := str(session.level.get("coach", ""))
	if key.is_empty() or (Progress.coach_seen(key) and not session.soft_mistakes):
		_set_coach("")
		return
	_set_coach(key)
	if key == "COACH_TAP" and session.board.alive_count == session.board.size():
		var free := MoveResolver.escapable_ids(session.board)
		if not free.is_empty():
			_board_view.show_hint(free[0]) # tutorial pointer; not a charged hint

func _new_attempt_id() -> String:
	return "%x-%s" % [int(Time.get_unix_time_from_system()), Crypto.new().generate_random_bytes(6).hex_encode()]

# --- input -------------------------------------------------------------------------------------

func _on_tapped(id: int) -> void:
	if _ended or _transitioning or _paused or _ad_busy or AdsManager.is_presenting():
		return
	var event := session.tap(id, Time.get_ticks_msec())
	match event["type"]:
		"escape":
			_board_view.play_escape(event)
			Feel.escape(event["combo"])
			if not (event["rotated"] as PackedInt32Array).is_empty():
				Feel.rotate()
			if event["combo"] in COMBO_MILESTONES:
				_board_view.combo(session.board.cell_of(id), event["combo"])
				Feel.combo(event["combo"])
			if _coach_key in ["COACH_TAP", "COACH_OPEN", "COACH_LOOK", "COACH_BLOCKED", "COACH_CHAIN", "COACH_HEARTS", "STUCK_NOTICE", "UNDO_NEEDED"]:
				_set_coach("")
			if event["cleared"]:
				_on_cleared()
				return
		"blocked":
			_board_view.play_blocked(event)
			Feel.bump(event["life_lost"])
			_hearts.set_lives(session.lives)
			if session.soft_mistakes:
				_set_coach("COACH_BLOCKED")
			if event["failed"]:
				_on_failed()
				return
		_:
			return
	_after_move()

func _on_preview_shown(_id: int) -> void:
	if mode == "switch" and _coach_key == "COACH_PREVIEW":
		Progress.mark_coach("COACH_PREVIEW")

func _after_move() -> void:
	_refresh_buttons()
	_save_attempt(false)
	if session.is_stuck():
		_show_stuck()

func _show_stuck() -> void:
	_set_coach("STUCK_NOTICE", "UNDO_FREE", _on_undo)

# --- tools -------------------------------------------------------------------------------------

func _refresh_buttons() -> void:
	if session == null:
		return
	var playing := session.status == PuzzleSession.Status.PLAYING and not _ended
	_hint_button.disabled = not playing
	if session.hints_left() > 0:
		_hint_button.set_badge(session.hints_left())
	elif AdsManager.rewarded_offer_state() != "unavailable":
		_hint_button.set_badge(-1, true, "+1")
	else:
		_hint_button.set_badge(-1)
	_undo_button.disabled = not (playing and session.can_undo())
	if session.has_undo_feature():
		if session.is_stuck():
			_undo_button.set_badge(-1, false)
			_undo_button.set_label("UNDO_FREE")
		else:
			_undo_button.set_label("UNDO")
			var charges := session.free_undos + session.bonus_undos
			if charges > 0:
				_undo_button.set_badge(charges)
			elif session.can_undo() and AdsManager.rewarded_offer_state() != "unavailable":
				_undo_button.set_badge(-1, true, "+1")
			else:
				_undo_button.set_badge(-1)
	_restart_button.disabled = _transitioning

func _on_hint() -> void:
	if session == null or _ended or _transitioning or _ad_busy:
		return
	var result := session.request_hint()
	match result["kind"]:
		"block":
			_board_view.show_hint(result["id"])
			Feel.hint()
			if result.get("charged", false):
				AnalyticsService.track(&"hint_used", _level_params({"source": "free" if session.bonus_hints == 0 and session.hints_used <= PuzzleSession.FREE_HINTS else "reward"}))
			_refresh_buttons()
			_save_attempt(false)
		"undo":
			_set_coach("UNDO_NEEDED", "UNDO", _on_undo)
		_:
			if str(result.get("reason", "")) == "no_hints":
				_offer_reward("hint")

func _on_undo() -> void:
	if session == null or _ended or _transitioning or _ad_busy or not session.can_undo():
		return
	if not session.undo_is_free():
		_offer_reward("undo")
		return
	var event := session.undo()
	if event["type"] != "undo":
		return
	_board_view.clear_hint()
	_board_view.play_undo(event)
	Feel.undo()
	AnalyticsService.track(&"undo_used", _level_params({"was_stuck": event["was_stuck"]}))
	if _coach_key in ["STUCK_NOTICE", "UNDO_NEEDED"]:
		_set_coach("")
	_after_move()

func restart_from_pause() -> void:
	_on_restart()

func _on_restart() -> void:
	if session == null or _transitioning or _ad_busy:
		return
	_track_level("level_restarted", {})
	_restart_level()

func _restart_level() -> void:
	var level := session.level
	var level_id := session.level_id()
	_flush_clock()
	_retries[level_id] = int(_retries.get(level_id, 0)) + 1
	Progress.clear_active()
	_begin_level(level, {}, true)

# --- rewarded ----------------------------------------------------------------------------------

func _offer_reward(kind: String) -> void:
	var state := AdsManager.rewarded_offer_state()
	if state == "unavailable":
		UI.toast("AD_UNAVAILABLE")
		return
	if state != "ready":
		UI.toast("AD_LOADING")
		AdsManager.retry_rewarded()
		return
	var title: String = {"hint": "REWARD_HINT_TITLE", "undo": "REWARD_UNDO_TITLE", "continue": "REWARD_CONTINUE_TITLE"}[kind]
	var body: String = {"hint": "REWARD_HINT_BODY", "undo": "REWARD_UNDO_BODY", "continue": "REWARD_CONTINUE_BODY"}[kind]
	UI.confirm(title, body, func(): _watch_reward(kind), "WATCH_AD")

func _watch_reward(kind: String) -> void:
	if session == null or _ad_busy:
		return
	var attempt := session.attempt_id
	_ad_busy = true
	_board_view.cancel_input()
	var started := AdsManager.show_rewarded(kind, attempt,
		func(transaction_id: String): _on_reward_earned(kind, attempt, transaction_id),
		func(result: Dictionary): _on_reward_closed(kind, attempt, result))
	if not started:
		_ad_busy = false

## Called once, from the SDK's earned-reward callback, possibly while the ad is still on screen.
## The grant is applied to the attempt it was earned for and saved immediately.
func _on_reward_earned(kind: String, attempt: String, transaction_id: String) -> void:
	if session == null or session.attempt_id != attempt:
		Diagnostics.log_event("ads", "reward_orphaned", {"kind": kind, "attempt_id": attempt, "tx": transaction_id})
		return
	if session.applied_rewards.has(transaction_id):
		return
	session.applied_rewards.append(transaction_id)
	match kind:
		"hint": session.grant_bonus_hint()
		"undo": session.grant_bonus_undo()
		"continue": session.continue_with_life()
	_save_attempt(true)

func _on_reward_closed(kind: String, attempt: String, result: Dictionary) -> void:
	_ad_busy = false
	if session == null or session.attempt_id != attempt:
		return
	if not bool(result.get("granted", false)):
		var reason := str(result.get("reason", ""))
		UI.toast("REWARD_NOT_EARNED" if reason == "dismissed_early" else "AD_UNAVAILABLE")
		_refresh_buttons()
		return
	match kind:
		"hint":
			_on_hint()
		"undo":
			_on_undo()
		"continue":
			_overlay.close()
			_ended = false
			_board_view.input_enabled = true
			_hearts.set_lives(session.lives)
			_set_coach("CONTINUE_NOTICE")
	_refresh_buttons()

# --- clear / fail ------------------------------------------------------------------------------

func _on_cleared() -> void:
	_ended = true
	_board_view.input_enabled = false
	_board_view.clear_hint()
	_flush_clock()
	var level_id := session.level_id()
	var record := Progress.record_clear(level_id, mode, session.is_perfect(), session.mistakes, session.daily_date)
	Progress.clear_active()
	var coach := str(session.level.get("coach", ""))
	if not coach.is_empty():
		Progress.mark_coach(coach)
	_track_level("level_completed", {"perfect": session.is_perfect(), "first_clear": record["first_clear"]})
	if mode == "daily" and record["first_clear"]:
		AnalyticsService.track(&"daily_challenge_completed", _level_params({"date": session.daily_date}))
	if level_id == "c005" and not Progress.flag("tutorial_completed"):
		Progress.set_flag("tutorial_completed")
		AnalyticsService.track(&"tutorial_completed", {"level_id": level_id})
	for theme_id in record["new_themes"]:
		AnalyticsService.track(&"theme_unlocked", {"theme": theme_id, "level_id": level_id})
	_refresh_buttons()
	var reduced := bool(SettingsManager.get_value("reduced_motion"))
	await get_tree().create_timer(0.08 if reduced else 0.3, false).timeout
	if not is_inside_tree():
		return
	_board_view.celebrate()
	Feel.clear()
	_set_coach("")
	await get_tree().create_timer(0.1 if reduced else 0.55, false).timeout
	if not is_inside_tree():
		return
	_show_clear_card(record)

func _show_clear_card(record: Dictionary) -> void:
	var body := _overlay.body
	_overlay.clear_content()
	var title := ""
	match mode:
		"switch": title = tr("SWITCH_CLEAR") % index
		"daily": title = tr("DAILY_CLEAR")
		_: title = tr("STAGE_CLEAR") % index
	var heading := Design.title(title, 30)
	heading.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	body.add_child(heading)
	var badges := HFlowContainer.new()
	badges.alignment = FlowContainer.ALIGNMENT_CENTER
	badges.add_theme_constant_override("h_separation", 8)
	badges.add_theme_constant_override("v_separation", 8)
	var p := Palette.colors()
	if session.is_perfect():
		var perfect := Design.row(6)
		perfect.add_child(Widgets.texture("star", 20, p["gold"]))
		perfect.add_child(Widgets.pill("BADGE_PERFECT", Color(p["gold"], 0.18), p["ink"], 16))
		badges.add_child(perfect)
	if mode == "campaign" and index % Levels.CHAPTER_SIZE == 0 and record["first_clear"]:
		badges.add_child(Widgets.pill(tr("CHAPTER_DONE") % Levels.chapter_of(index), p["accent_soft"], p["accent_deep"], 16))
	for theme_id in record["new_themes"]:
		badges.add_child(Widgets.pill(tr("THEME_UNLOCKED") % tr(Palette.THEMES[theme_id]["name"]), p["accent_soft"], p["accent_deep"], 16))
		Feel.unlock()
	if record["switch_unlocked"]:
		badges.add_child(Widgets.pill("SWITCH_UNLOCKED", p["accent_soft"], p["accent_deep"], 16))
	if badges.get_child_count() > 0:
		body.add_child(badges)
	var stats := Design.label(tr("CLEAR_STATS") % [session.board.size(), session.mistakes], 17, true, 600)
	stats.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	body.add_child(stats)
	body.add_child(Design.space(0.5))
	var next := _next_target()
	if not next.is_empty():
		var next_button := Design.button(next["label"], _on_next)
		next_button.icon = Widgets.icon("play")
		next_button.icon_alignment = HORIZONTAL_ALIGNMENT_RIGHT
		next_button.custom_minimum_size.y = 72
		body.add_child(next_button)
		next_button.grab_focus.call_deferred()
	else:
		var done := Design.label("ALL_DONE" if mode == "campaign" else ("SWITCH_ALL_DONE" if mode == "switch" else "DAILY_DONE_BODY"), 18, true)
		done.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
		body.add_child(done)
	body.add_child(Design.button("HOME", _go_home, true))
	_overlay.open()

## The next board after a clear, or {} when there is none (daily, last stage).
func _next_target() -> Dictionary:
	match mode:
		"campaign":
			if index < Levels.campaign_count():
				return {"label": "NEXT_STAGE", "mode": "campaign", "index": index + 1}
		"switch":
			if index < Levels.switch_count():
				return {"label": "NEXT_SWITCH", "mode": "switch", "index": index + 1}
	return {}

func _on_next() -> void:
	if _transitioning:
		return
	_transitioning = true
	_refresh_buttons()
	var next := _next_target()
	# Natural break: the ad (if one is ready and allowed) shows now or never. Continuation runs once.
	AdsManager.interstitial_opportunity("level_complete", {"level_index": index, "mode": mode},
		func(): _go_next(next))

func _go_next(next: Dictionary) -> void:
	if not is_inside_tree():
		return
	if next.is_empty():
		_go_home()
		return
	mode = next["mode"]
	index = next["index"]
	var level := Levels.switch_level(index) if mode == "switch" else Levels.campaign(index)
	_begin_level(level, {}, false)

func _on_failed() -> void:
	_ended = true
	_board_view.input_enabled = false
	_board_view.clear_hint()
	_flush_clock()
	_track_level("level_failed", {})
	_save_attempt(true)
	Feel.fail()
	await get_tree().create_timer(0.45, false).timeout
	if not is_inside_tree():
		return
	_show_fail_card()

func _show_fail_card() -> void:
	_ended = true
	_board_view.input_enabled = false
	var body := _overlay.body
	_overlay.clear_content()
	var heading := Design.title("OUT_OF_HEARTS", 28)
	heading.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	body.add_child(heading)
	var reason := Design.label("FAIL_REASON", 18, true, 600)
	reason.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	body.add_child(reason)
	body.add_child(Design.space(0.5))
	var restart := Design.button("RESTART_FREE", _on_restart)
	restart.icon = Widgets.icon("restart")
	restart.custom_minimum_size.y = 72
	body.add_child(restart)
	_continue_button = null
	_continue_note = null
	if session.can_continue() and AdsManager.rewarded_offer_state() != "unavailable":
		_continue_button = Design.button("CONTINUE_AD", func(): _offer_reward("continue"), true)
		_continue_button.icon = Widgets.icon("ad")
		body.add_child(_continue_button)
		_continue_note = Design.label("", 15, true, 600)
		_continue_note.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
		body.add_child(_continue_note)
		_refresh_continue()
	body.add_child(Design.button("HOME", _go_home, true))
	restart.grab_focus.call_deferred()
	_overlay.open()

func _refresh_continue() -> void:
	if _continue_button == null or not is_instance_valid(_continue_button):
		return
	var state := AdsManager.rewarded_offer_state()
	_continue_button.disabled = state != "ready"
	_continue_note.text = "" if state == "ready" else ("AD_LOADING" if state == "loading" else "AD_RETRY_LATER")
	_continue_note.visible = state != "ready"

# --- navigation & back -------------------------------------------------------------------------

func _go_home() -> void:
	if _ad_busy:
		return
	_flush_clock()
	_save_attempt(true)
	AppManager.return_to_menu()

## Android back inside the game: cards go home; otherwise the base pauses.
func _handle_back() -> bool:
	if _ad_busy:
		return true
	if _overlay.is_open():
		_go_home()
		return true
	return false

# --- persistence / clock / analytics -----------------------------------------------------------

func _save_attempt(immediate: bool) -> void:
	if session == null or session.status == PuzzleSession.Status.CLEARED:
		return
	var data := session.to_save()
	if immediate:
		Progress.save_active_now(data)
	else:
		Progress.stage_active(data)

func _process(delta: float) -> void:
	if session == null or _ended or _paused or AdsManager.is_presenting() or session.status != PuzzleSession.Status.PLAYING:
		return
	_clock_ms += delta * 1000.0
	if _clock_ms >= CLOCK_FLUSH_MS:
		_flush_clock()

func _flush_clock() -> void:
	if session == null or _clock_ms < 1.0:
		return
	var chunk := int(_clock_ms)
	_clock_ms -= chunk
	session.active_ms += chunk
	Progress.add_play_ms(chunk)

func _level_params(extra: Dictionary) -> Dictionary:
	var params := {"level_id": session.level_id(), "ruleset": str(session.level.get("rules", "")),
		"seed": int(session.level.get("seed", 0)), "attempt_id": session.attempt_id, "mode": mode,
		"active_play_time": session.active_ms, "mistakes": session.mistakes, "hints_used": session.hints_used,
		"retry_count": session.retry_count}
	params.merge(extra, true)
	return params

func _track_level(event: String, extra: Dictionary) -> void:
	AnalyticsService.track(StringName(event), _level_params(extra))
